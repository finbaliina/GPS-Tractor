# Straight contour-ploughing planner

# Search headings this finely.
# 1 degree is a good starting point.
CONTOUR_STRAIGHT_ANGLE_STEP_DEG=1.0

# How strongly the program prefers following terrain contours rather than
# simply choosing the shortest / fewest-turn route.
#
# Higher = contour alignment matters more.
# Lower = route efficiency matters more.
CONTOUR_ALIGNMENT_WEIGHT=8.0

# Ignore terrain cells flatter than this when calculating contour direction.
# Very flat cells have a noisy / meaningless slope direction.
CONTOUR_DIRECTION_MIN_SLOPE_DEG=1.0

# Additional overlap used in contour mode.
CONTOUR_EXTRA_OVERLAP=0.08
